cmake_minimum_required(VERSION 3.25)
project(HoroReference LANGUAGES C)
find_package(HoroEngineExtensionSdk CONFIG REQUIRED)
if(HORO_EXTENSION_ABI_PUBLISHED_MINOR LESS 8)
    message(FATAL_ERROR "These examples require extension host ABI1.8")
endif()
include(CTest)
set(EXAMPLE_ID "com.horo.examples.hybrid-basic" CACHE STRING "Canonical package identity; set once when copying")
if(NOT EXAMPLE_ID MATCHES "^[a-z][a-z0-9-]*(\\.[a-z][a-z0-9-]*)+$")
    message(FATAL_ERROR "EXAMPLE_ID must be a lowercase dotted identity")
endif()
configure_file(ExampleConfig.h.in generated/ExampleConfig.h @ONLY)
configure_file(extension.json.in extension.json @ONLY)
configure_file(horo-package.toml.in horo-package.toml @ONLY)

function(reference_target target role)
    target_compile_features(${target} PRIVATE c_std_11)
    target_link_libraries(${target} PRIVATE HoroEngine::ExtensionSdk)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    set(required "HORO_EXTENSION_REQUIRES_SERVICES")
    if(role STREQUAL "editor")
        set(required "HORO_EXTENSION_REQUIRES_EDITOR_ACTIVITY")
        if(1)
            string(APPEND required "|HORO_EXTENSION_REQUIRES_SERVICES")
        endif()
    endif()
    target_compile_definitions(${target} PRIVATE
        "EXAMPLE_ROLE=\"${role}\"" "EXAMPLE_REQUIRED_FUNCTIONS=(${required})"
        EXAMPLE_HAS_BACKEND=1 EXAMPLE_HAS_SCRIPT=1)
    set_target_properties(${target} PROPERTIES C_EXTENSIONS OFF)
endfunction()

foreach(role IN ITEMS backend script editor)
    add_library(reference_${role} MODULE src/${role}.c)
    reference_target(reference_${role} "${role}")
    set_target_properties(reference_${role} PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden)
    install(TARGETS reference_${role} DESTINATION bin)
    if(BUILD_TESTING)
        add_executable(reference_${role}_contract tests/contract.c src/${role}.c)
        reference_target(reference_${role}_contract "${role}")
        target_compile_definitions(reference_${role}_contract PRIVATE
            "EXAMPLE_IS_BACKEND=$<STREQUAL:${role},backend>"
            "EXAMPLE_IS_EDITOR=$<STREQUAL:${role},editor>")
        add_test(NAME reference_${role}_contract COMMAND reference_${role}_contract)
    endif()
endforeach()
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/extension.json" "${CMAKE_CURRENT_BINARY_DIR}/horo-package.toml" DESTINATION .)
install(DIRECTORY resources/ DESTINATION resources)
set(CPACK_PACKAGE_NAME "${EXAMPLE_ID}")
set(CPACK_PACKAGE_VERSION "1.0.0")
set(CPACK_PACKAGE_FILE_NAME "${EXAMPLE_ID}-1.0.0")
set(CPACK_GENERATOR ZIP)
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
include(CPack)
