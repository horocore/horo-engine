cmake_minimum_required(VERSION 3.25)
project(NetworkChaosProductBoundary LANGUAGES NONE)
list(APPEND CMAKE_MODULE_PATH "${HORO_ENGINE_SOURCE_DIR}/cmake")
include(HoroNetworkChaosBoundary)
include(HoroNetworkTransportComposition)
if(HORO_CASE STREQUAL "inventory")
    set(BUILD_TESTING OFF)
    add_library(HoroNetworkTransportNull INTERFACE)
    horo_verify_network_chaos_release_inventory()
elseif(HORO_CASE STREQUAL "headers")
    set(BUILD_TESTING OFF)
    add_library(leak INTERFACE)
    set_property(GLOBAL PROPERTY HORO_HEADER_BOUNDARY_TARGETS leak)
    set_property(TARGET leak PROPERTY HORO_PUBLIC_HEADERS Horo/Network/NetworkChaosScenario.h)
    horo_verify_network_chaos_release_inventory()
elseif(HORO_CASE STREQUAL "profile")
    set(BUILD_TESTING OFF)
    add_library(consumer INTERFACE)
    horo_configure_network_transport_profile(consumer HEADLESS_NULL)
else()
    add_library(HoroNetworkTransportNull INTERFACE)
    add_library(HoroEngine::NetworkTransportNull ALIAS HoroNetworkTransportNull)
    add_library(HoroNetworkChaosHarness INTERFACE)
    target_link_libraries(HoroNetworkChaosHarness INTERFACE HoroEngine::NetworkTransportNull)
    add_library(${HORO_HOST} INTERFACE)
    target_link_libraries(${HORO_HOST} INTERFACE "$<$<CONFIG:Release>:HoroNetworkChaosHarness>")
    horo_verify_network_chaos_boundary(${HORO_HOST})
endif()
