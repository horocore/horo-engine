cmake_minimum_required(VERSION 3.29)
project(HoroCliHeadlessPolicyContract NONE)
list(APPEND CMAKE_MODULE_PATH "${HORO_ENGINE_SOURCE_DIR}/cmake")
include(HoroCliHeadlessPolicy)
if(NOT HORO_TEST_ROOT)
    set(HORO_TEST_ROOT host)
endif()
add_library(${HORO_TEST_ROOT} INTERFACE)
add_library(application INTERFACE)
add_library(HoroFoundation INTERFACE)
add_library(HoroRenderApi INTERFACE)
add_library(HoroRenderNull INTERFACE)
target_link_libraries(${HORO_TEST_ROOT} INTERFACE application)
target_link_libraries(application INTERFACE HoroFoundation)
if(NOT HORO_TEST_ROOT STREQUAL "horopak")
    target_link_libraries(application INTERFACE HoroRenderApi HoroRenderNull)
endif()
if(HORO_FORBIDDEN_TARGET)
    if(NOT TARGET "${HORO_FORBIDDEN_TARGET}")
        if(HORO_FORBIDDEN_TARGET MATCHES "::")
            add_library("${HORO_FORBIDDEN_TARGET}" INTERFACE IMPORTED)
        else()
            add_library("${HORO_FORBIDDEN_TARGET}" INTERFACE)
        endif()
    endif()
    add_library(Forbidden::Alias ALIAS "${HORO_FORBIDDEN_TARGET}")
    target_link_libraries(application INTERFACE "$<LINK_ONLY:Forbidden::Alias>")
endif()
horo_verify_cli_headless_dependencies(${HORO_TEST_ROOT})
