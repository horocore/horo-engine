cmake_minimum_required(VERSION 3.25)
project(HoroSaveObservabilityDependencyFixture LANGUAGES NONE)
if(NOT HORO_ENGINE_SOURCE_DIR)
    message(FATAL_ERROR "HORO_ENGINE_SOURCE_DIR is required")
endif()
list(APPEND CMAKE_MODULE_PATH "${HORO_ENGINE_SOURCE_DIR}/cmake")
include(HoroDependencyDirection)
include(HoroDependencyPolicy)
foreach(target HoroFoundation HoroSaveApi HoroRuntime HoroApplication horo-engine)
    add_library(${target} INTERFACE)
endforeach()
target_link_libraries(HoroSaveApi INTERFACE HoroFoundation)
target_link_libraries(HoroRuntime INTERFACE HoroFoundation HoroSaveApi)
target_link_libraries(HoroApplication INTERFACE HoroFoundation)
target_link_libraries(horo-engine INTERFACE HoroApplication HoroRuntime)
if(HORO_REVERSE_OWNER)
    target_link_libraries(${HORO_REVERSE_OWNER} INTERFACE HoroRuntime)
endif()
horo_verify_dependency_direction(TARGETS HoroFoundation HoroSaveApi HoroRuntime HoroApplication horo-engine)
