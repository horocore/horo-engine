cmake_minimum_required(VERSION 3.25)
if(HORO_DEPENDENCY_CASE)
    project(HoroSecurityEnvelopeDirectionFixture LANGUAGES NONE)
    list(APPEND CMAKE_MODULE_PATH "${HORO_ENGINE_SOURCE_DIR}/cmake")
    include(HoroDependencyDirection)
    foreach(target HoroFoundation HoroSecurityEnvelope HoroSecurity HoroPackages HoroPackageSecurity horo-engine)
        add_library(${target} INTERFACE)
    endforeach()
    target_link_libraries(HoroSecurityEnvelope INTERFACE HoroFoundation)
    target_link_libraries(HoroSecurity INTERFACE HoroFoundation HoroSecurityEnvelope)
    target_link_libraries(HoroPackages INTERFACE HoroFoundation HoroSecurityEnvelope)
    if(HORO_DEPENDENCY_CASE STREQUAL "provider")
        target_link_libraries(HoroPackages INTERFACE HoroSecurity)
    elseif(HORO_DEPENDENCY_CASE STREQUAL "package-security")
        target_link_libraries(HoroPackages INTERFACE HoroPackageSecurity)
    elseif(HORO_DEPENDENCY_CASE STREQUAL "app")
        target_link_libraries(HoroPackages INTERFACE horo-engine)
    elseif(NOT HORO_DEPENDENCY_CASE STREQUAL "envelope")
        message(FATAL_ERROR "Unknown dependency case")
    endif()
    include(HoroDependencyPolicy)
    horo_verify_dependency_direction(TARGETS HoroFoundation HoroSecurityEnvelope HoroSecurity HoroPackages)
    return()
endif()
