cmake_minimum_required(VERSION 3.25)
project(ProtobufCoverageScope NONE)
# Configure-only compiler models: this fixture must never be built.
set(CMAKE_CXX_CREATE_STATIC_LIBRARY "scope-contract-not-built")
set(CMAKE_CXX_COMPILER_ID "${HORO_MODEL_COMPILER}")
set(CMAKE_CXX_COMPILER_VERSION "${HORO_MODEL_VERSION}")
set(HORO_CI_SONAR_COVERAGE "${HORO_MODEL_COVERAGE}")
set(CMAKE_CXX_FLAGS_DEBUG "--coverage -fprofile-abs-path")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/dummy.cc" "int value() { return 1; }\n")
foreach(target libprotobuf libprotobuf-lite product)
    add_library(${target} STATIC "${CMAKE_CURRENT_BINARY_DIR}/dummy.cc")
    set_target_properties(${target} PROPERTIES LINKER_LANGUAGE CXX)
endforeach()
add_library(protobuf::libprotobuf ALIAS libprotobuf)
target_link_libraries(product PRIVATE protobuf::libprotobuf)
include("${HORO_ENGINE_SOURCE_DIR}/cmake/HoroProtobufCoverage.cmake")
foreach(target libprotobuf libprotobuf-lite product)
    get_target_property(options ${target} COMPILE_OPTIONS)
    get_target_property(interface ${target} INTERFACE_COMPILE_OPTIONS)
    set(expected "options-NOTFOUND")
    if(HORO_MODEL_EXCLUDED AND NOT target STREQUAL "product")
        set(expected "-fprofile-exclude-files=.*")
    endif()
    if(NOT "${options}" STREQUAL "${expected}")
        message(FATAL_ERROR "${target}: COMPILE_OPTIONS '${options}', expected '${expected}'")
    endif()
    if(NOT "${interface}" STREQUAL "interface-NOTFOUND")
        message(FATAL_ERROR "${target}: coverage options leaked to its interface: ${interface}")
    endif()
endforeach()
if(NOT CMAKE_CXX_FLAGS_DEBUG STREQUAL "--coverage -fprofile-abs-path")
    message(FATAL_ERROR "Product coverage flags were changed: ${CMAKE_CXX_FLAGS_DEBUG}")
endif()
